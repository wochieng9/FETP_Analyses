% =====================================================================
%  Figure 1. Stock-and-flow structure of the African field epidemiology
%  workforce model.
%
%  Compiles on its own:   pdflatex fig1_stock_and_flow_tikz.tex
%  Produces a tight-cropped PDF suitable for \includegraphics.
%
%  To drop into a manuscript instead, either
%     \includegraphics[width=\textwidth]{fig1_stock_and_flow_tikz.pdf}
%  or copy the tikzpicture environment and add to your preamble:
%     \usepackage{tikz}
%     \usetikzlibrary{arrows.meta,positioning,shapes.symbols,calc,fit,
%                     backgrounds,decorations.markings}
%
%  Notation follows Sterman (2000, ch. 6): rectangles are stocks, thick
%  arrows with a bow-tie valve are material flows, clouds are sources and
%  sinks outside the model boundary, thin arrows are information links with
%  + / - polarity, // marks a delay, and circled R / B mark reinforcing and
%  balancing loops.
% =====================================================================

% bow-tie valve on a flow
\tikzset{
  pics/valve/.style={code={
    \filldraw[fill=white, draw=#1, line width=.75pt]
      (0,0) -- (-1.7mm, 1.35mm) -- (-1.7mm,-1.35mm) -- cycle;
    \filldraw[fill=white, draw=#1, line width=.75pt]
      (0,0) -- ( 1.7mm, 1.35mm) -- ( 1.7mm,-1.35mm) -- cycle;
  }},
  pics/valve/.default=flowcol,
}

% circled loop identifier with a curved arrow inside
\tikzset{
  pics/loop/.style 2 args={code={
    \filldraw[fill=white, draw=#1, line width=1pt] (0,0) circle (3.4mm);
    \draw[#1, line width=.6pt, -{Latex[length=1.5mm,width=1.3mm]}]
      (70:2.3mm) arc (70:-230:2.3mm);
    \node[font=\sffamily\scriptsize\bfseries, text=#1] at (0,0) {#2};
  }},
}
